# Dropdown

Absolutely-positioned menus anchored to a trigger. Toggle visibility with `aria-hidden` or `.nf-hidden` on the menu.

## JavaScript

```js
document.querySelectorAll('.nf-dropdown').forEach((dropdown) => {
  const trigger = dropdown.querySelector('.nf-dropdown-trigger');
  const menu    = dropdown.querySelector('.nf-dropdown-menu');
  const items   = () => [...menu.querySelectorAll('.nf-dropdown-item:not([aria-disabled="true"]):not(:disabled)')];

  // The CSS hides the menu only for aria-hidden="true" (or .nf-hidden),
  // so always write the literal string rather than toggling the attribute.
  function setOpen(open, { focusTrigger = false, focusItem = open } = {}) {
    trigger.setAttribute('aria-expanded', String(open));
    menu.setAttribute('aria-hidden', String(!open));
    if (open && focusItem) items()[0]?.focus();
    else if (focusTrigger) trigger.focus();
  }

  // Synchronize initial ARIA state without moving focus on page load.
  setOpen(trigger.getAttribute('aria-expanded') === 'true', { focusItem: false });

  trigger.addEventListener('click', (e) => {
    e.stopPropagation();
    setOpen(trigger.getAttribute('aria-expanded') !== 'true');
  });

  // Arrow keys move between items; Escape closes and returns focus.
  menu.addEventListener('keydown', (e) => {
    const list = items();
    const i = list.indexOf(document.activeElement);
    if (e.key === 'ArrowDown') { e.preventDefault(); list[(i + 1) % list.length]?.focus(); }
    if (e.key === 'ArrowUp')   { e.preventDefault(); list[(i - 1 + list.length) % list.length]?.focus(); }
    if (e.key === 'Escape')    { setOpen(false, { focusTrigger: true }); }
  });

  // Close on outside click
  document.addEventListener('click', (e) => {
    if (!dropdown.contains(e.target)) setOpen(false);
  });
});
```

## Class reference

| Class | Description |
|---|---|
| `nf-dropdown` | `position: relative` wrapper |
| `nf-dropdown-trigger` | Button with caret indicator |
| `nf-dropdown-menu` | Absolute menu panel — hide with `aria-hidden="true"` or `nf-hidden` |
| `nf-dropdown-menu-right` | Right-aligns the menu |
| `nf-dropdown-item` | Menu row — button or anchor |
| `nf-dropdown-item-danger` | Error color variant |
| `nf-dropdown-divider` | `<li>` separator between item groups |
| `nf-dropdown-label` | `<li>` group header — non-interactive |
