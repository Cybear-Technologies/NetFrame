# Tabs

## Many tabs on narrow screens

The tab list scrolls horizontally inside itself (the page never does). The JavaScript example keeps the selected tab in view, including a tab that starts off-screen. The scrollbar is hidden by design, so avoid more tabs than a reader can discover by swiping; consider an accordion or a select for long lists.

## JavaScript

```js
document.querySelectorAll('.nf-tab-list').forEach((list) => {
  const tabs = [...list.querySelectorAll('.nf-tab')];
  const enabled = () => tabs.filter((t) => t.getAttribute('aria-disabled') !== 'true' && !t.disabled);

  function select(tab, { focus = false } = {}) {
    if (!tab || tab.getAttribute('aria-disabled') === 'true' || tab.disabled) return;
    tabs.forEach((t) => {
      const active = t === tab;
      t.setAttribute('aria-selected', String(active));
      t.tabIndex = active ? 0 : -1; // roving tabindex: one tab stop per list
      document.getElementById(t.getAttribute('aria-controls'))?.toggleAttribute('hidden', !active);
    });
    revealSelected();
    if (focus) tab.focus();
  }

  function revealSelected() {
    const tab = tabs.find((t) => t.getAttribute('aria-selected') === 'true');
    if (!tab) return;
    // Scroll only the list, never the page. Font swaps and container resizing
    // can change these bounds after the initial selection.
    const a = tab.getBoundingClientRect(), b = list.getBoundingClientRect();
    if (a.left < b.left) list.scrollLeft -= b.left - a.left;
    else if (a.right > b.right) list.scrollLeft += a.right - b.right;
  }

  select(enabled().find((t) => t.getAttribute('aria-selected') === 'true') ?? enabled()[0]);
  const resize = new ResizeObserver(revealSelected);
  resize.observe(list);
  tabs.forEach((tab) => resize.observe(tab));

  list.addEventListener('click', (e) => {
    const tab = e.target.closest('.nf-tab');
    if (tab) select(tab);
  });

  // Arrow keys / Home / End move between enabled tabs (automatic activation).
  list.addEventListener('keydown', (e) => {
    const order = enabled();
    const i = order.indexOf(document.activeElement);
    const next = {
      ArrowRight: order[(i + 1) % order.length],
      ArrowLeft: order[(i - 1 + order.length) % order.length],
      Home: order[0],
      End: order[order.length - 1],
    }[e.key];
    if (next) { e.preventDefault(); select(next, { focus: true }); }
  });
});
```

## Class reference

| Class | Description |
|---|---|
| `nf-tabs` | Outer wrapper |
| `nf-tab-list` | Horizontal row of tab buttons |
| `nf-tab` | Individual tab button — set `aria-selected="true"` for active |
| `nf-tab-active` | Class alternative to `aria-selected` for active state |
| `nf-tabs-secondary` | Secondary magenta active underline |
| `nf-tab-panel` | Content panel — hide inactive panels with `hidden` |
